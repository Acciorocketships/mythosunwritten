## Closed, wall-embedded crags with independently graded, broken turf ledges.
## Detached geometry is shared by rendering, collision, grass and crevice plants.

static func make(panel:Transform3D,width:float,height:float,seed_value:int,region:HeightfieldRegion=null)->Array[Dictionary]:
 var result:Array[Dictionary]=[]
 var count:=maxi(1,roundi(width/6.5))
 var bay:=width/count
 for i in count:
  var anchor:Vector3=panel*Vector3(-width*.5+bay*(i+.5),0,0)
  var rng:=RandomNumberGenerator.new();rng.seed=hash([anchor,seed_value,18737])
  var span:=bay*rng.randf_range(1.10,1.55)
  var tall:=height*rng.randf_range(.45,.96)
  var levels:=maxi(2,ceili(tall/3.5)+rng.randi_range(0,1))
  var floor_y:=-.18
  if region!=null:
   for x in range(-ceili(span*.5),ceili(span*.5)+1):
    for z in [0.0,1.5,3.5]:
     var p:Vector3=panel*Vector3(-width*.5+bay*(i+.5)+x,0,z)
     floor_y=minf(floor_y,TerrainSurfaceField.surface_y(region,p.x,p.z)-panel.origin.y-.15)
  var heights:Array[float]=[];var depths:Array[float]=[];var widths:Array[float]=[];var slopes:Array[float]=[];var shifts:Array[float]=[]
  var total:=0.0
  for level in levels:
   total+=rng.randf_range(.5,1.6);heights.append(total)
   depths.append(rng.randf_range(1.3,4.2));widths.append(rng.randf_range(.45,1.15));slopes.append(rng.randf_range(-.22,.22));shifts.append(rng.randf_range(-.18,.18)*span)
  for level in levels:heights[level]=lerpf(floor_y,tall,heights[level]/total)
  # Broad lower bearing, without reverse-tapered hanging shelves.
  for level in range(levels-2,-1,-1):depths[level]=minf(4.4,maxf(depths[level],depths[level+1]+rng.randf_range(.15,.55)))
  var nx:=maxi(8,ceili(span/1.1));var columns:Array[PackedVector3Array]=[]
  var base_x:float=-width*.5+bay*(i+.5)
  for ix in nx+1:
   var x:=span*(float(ix)/nx-.5)
   var base_shoulder:=clampf((span*.5*widths[0]-absf(x-shifts[0]))/(span*.10),0.0,1.0)
   var column:=PackedVector3Array([Vector3(base_x+x,floor_y,lerpf(-1.0,depths[0]+.15,base_shoulder))])
   var last_top:=floor_y
   for level in levels:
    var top:float=heights[level]+slopes[level]*x+.19*sin(x*1.4+level*2.7+i)
    top=clampf(top,last_top+.38,height-.25-(levels-level-1)*.4)
    var thickness:=maxf(.12,(top-last_top)*.80)
    last_top=top+.08
    var shoulder:=clampf((span*.5*widths[level]-absf(x-shifts[level]))/(span*.10),0.0,1.0)
    var depth:float=lerpf(-1.0,depths[level]+.30*sin(x*1.7+level*3.3)+.12*sin(x*3.4+1.9),shoulder)
    var retreat:float=maxf(-1.0,depth-(.8+depths[level]*.35))
    column.append(Vector3(base_x+x,top-thickness,depth-.20*shoulder))
    column.append(Vector3(base_x+x+.12*sin(x+level),top-thickness*.63,depth+shoulder*.22*sin(x*1.6+level*1.7)))
    column.append(Vector3(base_x+x-.10*sin(x*1.3+level),top-thickness*.30,depth+shoulder*.27*sin(x*1.2+level*4.7)))
    column.append(Vector3(base_x+x,top-.16,depth))
    column.append(Vector3(base_x+x,top,depth-.15*shoulder))
    column.append(Vector3(base_x+x,top+.08+slopes[level]*.6,retreat))
   column.append(Vector3(base_x+x,minf(height-.02,maxf(tall+.3,last_top+.1)),-1.0))
   columns.append(column)
  var faces:=PackedVector3Array();var green:=PackedVector3Array()
  for ix in nx:
   for row in columns[ix].size()-1:
    var a:Vector3=columns[ix][row];var b:Vector3=columns[ix+1][row];var c:Vector3=columns[ix][row+1];var d:Vector3=columns[ix+1][row+1]
    for tri:PackedVector3Array in [PackedVector3Array([a,c,b]),PackedVector3Array([b,c,d])]:
     faces.append_array(tri)
     var n:Vector3=(tri[2]-tri[0]).cross(tri[1]-tri[0]).normalized()
     if row%6==5 and n.y>.55 and minf(tri[0].z,minf(tri[1].z,tri[2].z))>.15:green.append_array(tri)
  var perimeter:=PackedVector3Array()
  for col:PackedVector3Array in columns:perimeter.append(col[0])
  for row in range(1,columns[-1].size()):perimeter.append(columns[-1][row])
  for ix in range(nx-1,-1,-1):perimeter.append(columns[ix][-1])
  for row in range(columns[0].size()-2,0,-1):perimeter.append(columns[0][row])
  var center:=Vector3(base_x,(floor_y+tall)*.5,-1.5)
  for edge in perimeter.size():
   var a:Vector3=perimeter[edge];var b:Vector3=perimeter[(edge+1)%perimeter.size()]
   faces.append_array(PackedVector3Array([a,b,center]))
  var bounds:=AABB(faces[0],Vector3.ZERO)
  for point:Vector3 in faces:bounds=bounds.expand(point)
  result.append({"faces":faces,"green":green,"bounds":panel*bounds,"transform":panel,"anchor":anchor,
   "id":"crag/%s"%anchor,"asset":&"cliff.native_crag","kind":"rock","native_crag":true,"top":(panel*bounds).end.y,"base":(panel*bounds).position.y})
 return result

static func mesh(rock:Dictionary)->ArrayMesh:
 var result:=ArrayMesh.new();var turf:Dictionary={};var adjacent:Dictionary={}
 for i in range(0,rock.faces.size(),3):
  var a:Vector3=rock.faces[i];var b:Vector3=rock.faces[i+1];var c:Vector3=rock.faces[i+2]
  var n:Vector3=(c-a).cross(b-a).normalized()
  for p:Vector3 in [a,b,c]:
   if not adjacent.has(p):adjacent[p]=[]
   adjacent[p].append(n)
 for i in range(0,rock.green.size(),3):turf[[rock.green[i],rock.green[i+1],rock.green[i+2]]]=true
 for grass in [false,true]:
  if grass and turf.is_empty():continue
  var st:=SurfaceTool.new();st.begin(Mesh.PRIMITIVE_TRIANGLES)
  for i in range(0,rock.faces.size(),3):
   var a:Vector3=rock.faces[i];var b:Vector3=rock.faces[i+1];var c:Vector3=rock.faces[i+2]
   if turf.has([a,b,c])!=grass:continue
   var normal:Vector3=(c-a).cross(b-a).normalized()
   for p:Vector3 in [a,b,c]:
    var average:=Vector3.ZERO
    for n:Vector3 in adjacent[p]:
     if n.dot(normal)>.89:average+=n
    st.set_normal(average.normalized());st.set_uv(CliffDressing.ground_uv());st.add_vertex(p)
  st.commit(result)
  if grass:result.surface_set_material(result.get_surface_count()-1,CliffDressing.shared_material())
  else:
   var mat:=ShaderMaterial.new();mat.shader=load("res://terrain/materials/field_rock.gdshader")
   mat.set_shader_parameter("use_texture",false);mat.set_shader_parameter("instance_variation",false)
   result.surface_set_material(0,mat)
 return result
