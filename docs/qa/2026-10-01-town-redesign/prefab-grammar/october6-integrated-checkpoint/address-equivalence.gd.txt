extends SceneTree
const Previous = preload("/tmp/oct6-crown-planner.gd")
const Candidate = preload("/tmp/oct6-fast-planner.gd")
func _init(): call_deferred("_run")
func _run():
 var results := {}
 for job: String in ["31:large", "53:grand", "103:grand"]:
  var fields := job.split(":")
  var versions := []
  var times := []
  for planner in [Previous, Candidate]:
   var started := Time.get_ticks_msec()
   var plan: WarrenMazeSourcePlan = planner.plan(int(fields[0]), {}, WarrenVillageScaleProfile.for_id(StringName(fields[1])))
   assert(plan != null)
   times.append(Time.get_ticks_msec()-started)
   var state := {"plots":plan.plots,"passages":plan.passage_kinds,"market":plan.market_square_cells,"features":plan.feature_stamps,"summit":plan.summit_cell}
   for field: String in preload("/tmp/oct6-short-address.gd").COPY_FIELDS:
    state[field] = plan.excavation.get(field)
   versions.append(var_to_str(state))
  results[job] = {"same_layout":versions[0] == versions[1],"previous_ms":times[0],"candidate_ms":times[1]}
  print("EQUIVALENCE ",job," ",JSON.stringify(results[job]))
  FileAccess.open("/tmp/oct6-address-equivalence.json",FileAccess.WRITE).store_string(JSON.stringify(results,"  "))
 quit()
