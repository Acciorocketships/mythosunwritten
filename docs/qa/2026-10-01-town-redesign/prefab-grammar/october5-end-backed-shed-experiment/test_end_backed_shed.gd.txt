extends GutTest
func host() -> BuildingMass:
 var mass:=BuildingMass.new()
 mass.add_storey(0,BuildingMass.rect_cells(Rect2i(0,0,4,1)),BuildingMass.MATERIAL_TIMBER)
 for band in [2,4,6]:mass.add_storey(band,BuildingMass.rect_cells(Rect2i(2,0,2,1)),BuildingMass.MATERIAL_TIMBER)
 return mass
func test_short_wing_uses_whole_pitch_into_tall_end_wall():
 var designer:=BuildingDesigner.new(SuntailBuildingKit.create())
 var spec:=designer._backed_shed(host(),Rect2i(0,0,2,1),2)
 assert_false(spec.is_empty())
 if spec.is_empty():return
 assert_eq(spec.rect,Rect2i(0,0,4,1))
 assert_eq(spec.axis,1)
 assert_eq(spec.cut_at,2)
 assert_eq(spec.backing.size(),3)
func test_end_shed_needs_full_height_backing_and_preserves_doors():
 var designer:=BuildingDesigner.new(SuntailBuildingKit.create())
 var mass:=host()
 mass.storeys.pop_back()
 assert_true(designer._backed_shed(mass,Rect2i(0,0,2,1),2).is_empty())
 mass=host()
 mass.storeys[2].openings[BuildingMass.edge_key(Vector2i(2,0),2)]=BuildingMass.OPENING_DOOR
 assert_true(designer._backed_shed(mass,Rect2i(0,0,2,1),2).is_empty())
 designer.forbidden=func(cell:Vector2i,band:int)->bool:return cell==Vector2i(1,0) and band==3
 assert_true(designer._backed_shed(host(),Rect2i(0,0,2,1),2).is_empty())
