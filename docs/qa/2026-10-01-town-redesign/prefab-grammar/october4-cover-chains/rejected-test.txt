

func test_supported_cover_continues_original_house_through_adjacent_room() -> void:
	var plan := WarrenMazeSourcePlan.new(1,WarrenVillageScaleProfile.for_id(&"standard"),null,null)
	plan.plots = [
		{"id":&"house", "kind":WarrenMazeSourcePlan.PLOT_HOUSE,"cells":[Vector2i(2,0)],"floor":0,"top":8},
		{"id":&"over", "kind":WarrenMazeSourcePlan.PLOT_OVER,"cells":[Vector2i(1,0)],"floor":4,"top":8,"host":&"house"},
	]
	plan._rebuild_plot_columns()
	assert_eq(WarrenPlotPlanner._tunnel_host(plan,Vector2i.ZERO,2),Vector2i(0,4),"Continue the original parcel through its supported room")
	plan.plots[1].floor=6
	assert_eq(WarrenPlotPlanner._tunnel_host(plan,Vector2i.ZERO,2).x,-1,"A disconnected higher room is not a host connection")
	plan.plots[1].floor=4
	plan.plots[1].host=&"missing"
	assert_eq(WarrenPlotPlanner._tunnel_host(plan,Vector2i.ZERO,2).x,-1,"Never invent a host for an orphan cover")
