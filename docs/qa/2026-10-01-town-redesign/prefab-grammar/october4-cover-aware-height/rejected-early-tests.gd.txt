extends GutTest


func test_optional_house_roll_stays_within_the_planned_massif() -> void:
	var columns := {
		Vector2i.ZERO: {"base": 0, "top": 10, "plinth": 4},
		Vector2i.RIGHT: {"base": 0, "top": 12, "plinth": 4}
	}
	var massif := WarrenMassif.with_columns(1, columns, 12)
	var profile := WarrenVillageScaleProfile.for_id(&"grand")
	var exercised := 0
	for seed_value in range(1, 33):
		var plan := WarrenMazeSourcePlan.new(
			seed_value, profile, massif, WarrenExcavation.new(seed_value)
		)
		var cells: Array[Vector2i] = [Vector2i.ZERO, Vector2i.RIGHT]
		var building := {"floor": 4, "seed": Vector2i.ZERO, "door": Vector3i.ZERO, "cells": cells}
		var roof := WarrenPlotPlanner._rolled_seed_top(plan, building)
		assert_lte(
			roof,
			10 + WarrenBuildingParcel.ROOF_RESERVATION_BANDS,
			"Optional walls must not grow above the volume that streets bored through"
		)
		exercised += int(roof == 10 + WarrenBuildingParcel.ROOF_RESERVATION_BANDS)
	assert_gt(exercised, 0, "Keep the full planned height available")


func test_required_upper_street_and_room_keep_their_bearing_height() -> void:
	var columns := {}
	for x in range(-5, 6):
		for z in range(-5, 6):
			columns[Vector2i(x, z)] = {"base": 0, "top": 4}
	var massif := WarrenMassif.with_columns(1, columns, 4)
	var plan := WarrenMazeSourcePlan.new(
		1, WarrenVillageScaleProfile.for_id(&"grand"), massif, WarrenExcavation.new(1)
	)
	var cells: Array[Vector2i] = [Vector2i.ZERO]
	var building := {"floor": 0, "seed": Vector2i.ZERO, "door": Vector3i.ZERO, "cells": cells}
	assert_lte(WarrenPlotPlanner._rolled_seed_top(plan, building), 6)
	var street := WarrenPlotPlanner._building_top(plan, {Vector2i.ZERO: [8]}, {}, building)
	assert_eq(street.top, 8, "An existing upper street cannot lose its supporting house")
	assert_true(street.tiered)
	var carried := WarrenPlotPlanner._building_top(plan, {}, {Vector2i.ZERO: [8]}, building)
	assert_eq(carried.top, 8, "An existing upper room cannot lose its supporting house")


func test_neighboring_path_keeps_a_complete_overhead_room_budget() -> void:
	var massif := WarrenMassif.with_columns(1, {Vector2i.ZERO: {"base": 0, "top": 2}}, 2)
	var profile := WarrenVillageScaleProfile.for_id(&"grand")
	for seed_value in range(1, 33):
		var plan := WarrenMazeSourcePlan.new(
			seed_value, profile, massif, WarrenExcavation.new(seed_value)
		)
		var cells: Array[Vector2i] = [Vector2i.ZERO]
		var building := {"floor": 0, "seed": Vector2i.ZERO, "door": Vector3i.ZERO, "cells": cells}
		var raw := (
			WarrenBuildingParcel.ROOF_RESERVATION_BANDS
			+ (
				WarrenBuildingParcel.STOREY_BANDS
				* WarrenPlotPlanner._building_roll(
					plan,
					building,
					WarrenPlotPlanner.STOREY_SALT,
					profile.scaled(WarrenPlotPlanner.STOREY_BUDGET)
				)
			)
		)
		plan.passage_kinds[Vector3i(1, 2, 0)] = WarrenMazeSourcePlan.PASSAGE_SPINE
		var height := WarrenPlotPlanner._rolled_seed_top(plan, building)
		assert_gte(height, mini(raw, 8), "Path at 2 needs headroom to 4, a whole room and its roof")
		assert_lte(
			height, raw, "A cover reserve cannot invent extra storeys beyond the original roll"
		)
		plan.passage_kinds.clear()
		plan.passage_kinds[Vector3i(8, 2, 0)] = WarrenMazeSourcePlan.PASSAGE_SPINE
		assert_eq(
			WarrenPlotPlanner._rolled_seed_top(plan, building),
			4,
			"Distant paths do not raise this house"
		)
