extends GutTest

func test_natural_ground_is_a_foundation_without_an_underground_massif_cell() -> void:
	var massif := WarrenMassif.new(3)
	var excavation := WarrenExcavation.new(3)
	var left := Vector2i.ZERO
	var right := Vector2i.RIGHT
	for column in [left, right]:
		massif.columns[column] = {"base": 0, "plinth": 0, "top": 12}
	assert_false(WarrenMazeCarver._column_is_solid_at(massif, excavation, left, -1),
		"The above-ground massif intentionally has no underground voxel")
	assert_true(WarrenMazeCarver._bridge_foundation_is_direct(massif, excavation, {}, [left, right], 0, 6),
		"A complete house on natural ground is directly borne")
	assert_false(WarrenMazeCarver._bridge_foundation_is_direct(massif, excavation, {}, [left], -1, 6),
		"Recognizing terrain never permits a house inside the ground")
	excavation.carved[Vector3i(0, 2, 0)] = true
	assert_false(WarrenMazeCarver._bridge_foundation_is_direct(massif, excavation, {}, [left], 0, 6),
		"A passage still interrupts the endpoint's required complete room stack")
	excavation.carved.clear()
	assert_false(WarrenMazeCarver._bridge_foundation_is_direct(massif, excavation, {Vector3i(1, 3, 0):true}, [left, right], 0, 6))
	massif.columns[right].base = 1
	assert_false(WarrenMazeCarver._bridge_foundation_is_direct(massif, excavation, {}, [left, right], 0, 6),
		"Every column must be supported; a mixed-datum footprint still fails")
